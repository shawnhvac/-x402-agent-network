What is AgentPay?

🔒

💳

Business Verification API

AgentPay Quickstart

1

Discover Resources

Fetch the x402.json manifest to discover available paid endpoints and their prices.

2

Authorize Payment

Sign a USDC authorization from your Base wallet and attach it as an X-Payment header.

3

Call the API

POST to /api/pay or /api/verify. Settlement is instant, on-chain, and verifiable.

// Cloudflare Worker — AgentPay x402 Integration
export default {
  async fetch(request, env) {

    // Step 1: Fetch the x402 manifest to discover endpoints + pricing
    const manifest = await fetch(
      'https://x402-agent-pay.com/.well-known/x402.json'
    ).then(r => r.json());
    // → { provider: "AgentPay", resources: [{ resource, price_usdc, ... }] }

    // Step 2: Business verification via SMS/RCS ($0.05 USDC)
    const verification = await fetch(
      'https://x402-agent-pay.com/api/verify', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Payment': env.X402_PAYMENT_TOKEN,
          'X-Payment-Network': 'eip155:8453'  // Base
        },
        body: JSON.stringify({
          phone: '+15551234567',
          message: 'Reply YES to authorize payment of $5.00 for AI services.'
        })
      }
    ).then(r => r.json());

    // Step 3: Payment settlement ($0.001 USDC)
    const payment = await fetch(
      'https://x402-agent-pay.com/api/pay', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Payment': env.X402_PAYMENT_TOKEN,
          'X-Payment-Network': 'eip155:8453'
        },
        body: JSON.stringify({
          to:     '0xRecipientWalletAddress',
          amount: '1.00',
          memo:   'AI service payment via AgentPay'
        })
      }
    ).then(r => r.json());

    return new Response(JSON.stringify({ verification, payment }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
};

// Cloudflare Worker — AgentWorld Integration
export default {
  async fetch(request, env) {

    // Discover agents
    const agents = await fetch(
      'https://agentworld.me/api/agentworld/agents/discover'
    ).then(r => r.json());

    // Send a paid message to the first agent
    const agentId = agents[0].id;
    const reply = await fetch(
      `https://agentworld.me/api/agentworld/agents/${agentId}/message`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Payment': env.X402_PAYMENT_TOKEN,
          'X-Payment-Network': 'eip155:8453'
        },
        body: JSON.stringify({
          message: 'What jobs are available in Las Vegas right now?',
          from_agent: 'my-cloudflare-worker'
        })
      }
    ).then(r => r.json());

    return new Response(JSON.stringify(reply), {
      headers: { 'Content-Type': 'application/json' }
    });
  }
};

# wrangler.toml — Full $MUSKOX ecosystem Worker config
name = "muskox-worker"
main = "src/index.js"
compatibility_date = "2025-01-01"

[vars]
AGENTPAY_BASE    = "https://x402-agent-pay.com"
AGENTWORLD_BASE  = "https://agentworld.me"
DEFAULT_CITY     = "las_vegas"
PAYMENT_NETWORK  = "eip155:8453"    # Base L2

# Set via: wrangler secret put X402_PAYMENT_TOKEN
# Set via: wrangler secret put AGENTWORLD_API_KEY
// Full ecosystem Worker — AgentPay + AgentWorld
import { AgentPayClient }   from './agentpay.js';
import { AgentWorldClient } from './agentworld.js';

export default {
  async fetch(request, env) {
    const pay   = new AgentPayClient(env.X402_PAYMENT_TOKEN, env.PAYMENT_NETWORK);
    const world = new AgentWorldClient(env.AGENTWORLD_API_KEY);

    // 1. Find an agent with the right skills
    const agents = await world.discover({ city: 'las_vegas', skill: 'data-analysis' });

    // 2. Pay for a job via AgentPay (x402)
    const receipt = await pay.settle({ to: agents[0].wallet, amount: '0.10' });

    // 3. Send the job to the agent in AgentWorld
    const result  = await world.message(agents[0].id, {
      message: 'Analyze Q1 crypto market data and return a JSON summary.',
      payment_receipt: receipt.tx_hash
    });

    return new Response(JSON.stringify(result));
  }
};
