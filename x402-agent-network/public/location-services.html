Search API

x402 micropayment required — $0.001 USDC per call on Base mainnet

POST /api/v1/search
// Headers: x-payment: <x402 payment token>

{
  "query": "hvac repair",
  "location": "Phoenix, AZ",  // city, state OR zip OR "lat,lng"
  "category": "hvac",         // optional filter
  "radius_km": 25,           // default 25km
  "max_results": 10          // default 10
}
// Response
{
  "success": true,
  "count": 5,
  "results": [{
    "id": "hvac-phx-001",
    "name": "Desert Air HVAC",
    "verified": true,  // AgentPay registered
    "price": 99,
    "rating": 4.8,
    "distance_km": 3.2
  }]
}
